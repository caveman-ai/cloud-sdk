`github.com/caveman-ai/sdk-go` is a standalone Go client for Caveman Cloud.

## Install

```bash
go get github.com/caveman-ai/sdk-go
```

## Unified client

```go
package main

import (
	"context"
	"fmt"
	cavemancloud "github.com/caveman-ai/sdk-go"
)

func main() {
	cv, err := cavemancloud.NewCaveman(cavemancloud.CavemanOptions{}) // reads CAVEMAN_API_KEY
	if err != nil {
		panic(err)
	}
	ctx := context.Background()
	models, _ := cv.ListModels(ctx)
	drives, _ := cv.Drives.List(ctx)
	fmt.Println(len(models), len(drives))
}
```
